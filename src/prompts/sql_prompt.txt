Utilize the following SQL as a template for queries about events/meetings/etc.

SELECT
  e.title AS event_title,
  to_char(e.start_dt AT TIME ZONE e.timezone, 'Month DD YYYY, HH12:MI AM') AS event_start_dt,
  e.timezone AS event_timezone,
  e.description,
  e.venue_name,
  e.room_name,
  e.address,
  e.type,
  n.html AS event_notes_meeting_minutes,
  STRING_AGG(CONCAT('\n Name: ', p.first_name, ' ', p.last_name,
                    '\n user_id: ', p.user_id,
                    '\n is_speaker: ', cpl.is_speaker,
                    '\n email: ', p.email,
                    '\n Afilliation: ', p.affiliation,
                    '\n Contribution Title: ', c.title,
                    '\n Contribution Description: ', c.description,
                    '\n Contribution Start Time: ', t.start_dt,
                    '\n Contribution Duration: ', c.duration,
                    '\n Contribution Attachments: ', f.filename,
                    '\n Notes: ', cn.html),', ') AS contributors
FROM
  events.events e
LEFT JOIN
  events.contributions c ON e.id = c.event_id
LEFT JOIN
  events.notes n ON e.id = n.event_id
LEFT JOIN
  events.notes cn ON c.id = cn.contribution_id
LEFT JOIN
  events.contribution_person_links cpl ON c.id = cpl.contribution_id
LEFT JOIN
  events.timetable_entries t ON c.id = t.contribution_id
LEFT JOIN
  attachments.legacy_attachment_id_map lam ON c.id = lam.contribution_id::INTEGER
LEFT JOIN
  attachments.files f ON lam.attachment_id = f.id
LEFT JOIN
  events.persons p ON cpl.person_id = p.id
WHERE
    e.is_deleted = false
    AND n.is_deleted = false
    AND e.id=7
    AND n.contribution_id IS NULL 
GROUP BY
    e.id,
    n.html;


And for queries about materials/attachments/pdfs/slides/etc. use the following SQL as a template:

SELECT STRING_AGG(fl.storage_file_id, ', '), p.first_name, p.last_name, p.user_id, cpl.is_speaker, p.affiliation, p.email 
FROM
    attachments.folders f
JOIN 
    events.contributions c ON f.contribution_id = c.id
JOIN
    attachments.attachments a ON f.id = a.folder_id
JOIN
    attachments.files fl ON a.file_id = fl.id
LEFT JOIN
  events.contribution_person_links cpl ON c.id = cpl.contribution_id
LEFT JOIN
  events.persons p ON cpl.person_id = p.id
WHERE
    f.event_id = {event_id}
GROUP BY
    fl.id
